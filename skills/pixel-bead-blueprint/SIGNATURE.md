- 证书指纹（SHA256）：4A:56:E5:FA:FB:41:A2:00:F6:B1:E0:7E:88:68:AA:AE:16:13:EA:C3:E2:E1:38:E4:48:60:23:1D:1B:5D:01:24

```powershell
# 只还原到临时目录，不要写回被签目录
$out = Join-Path $env:TEMP 'pbb-signature-restore'
New-Item -ItemType Directory -Force -Path $out | Out-Null
$t = Get-Content SIGNATURE.md -Raw
$ms = [regex]::Matches($t,'(?s)<!-- (\w+):START -->(.*?)<!-- \1:END -->')
$blocks = @{}
foreach ($m in $ms) { $blocks[$m.Groups[1].Value] = $m.Groups[2].Value }
[IO.File]::WriteAllBytes((Join-Path $out 'skill.oms.sig'), [Convert]::FromBase64String(($blocks['SIGBASE64'] -replace '\s','')))

[IO.File]::WriteAllText((Join-Path $out 'cert.pem'), $pem)
```
