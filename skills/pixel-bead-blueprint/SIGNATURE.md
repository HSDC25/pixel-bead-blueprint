# 只还原到临时目录，不要写回被签目录
$out = Join-Path $env:TEMP 'pbb-signature-restore'

$t = Get-Content SIGNATURE.md -Raw

[IO.File]::WriteAllText((Join-Path $out 'cert.pem'), $pem)
